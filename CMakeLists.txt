cmake_minimum_required(VERSION 3.20)
project(D2R3DPlugins LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
foreach(plugin IN ITEMS 3dcam renderdistance)
  add_library(d2rl-${plugin} SHARED plugin/${plugin}.cpp plugin/plugin.rc)
  target_include_directories(d2rl-${plugin} PRIVATE ../PluginSDK/include)
  target_compile_options(d2rl-${plugin} PRIVATE /utf-8 /W3 /EHsc /permissive-)
  target_link_libraries(d2rl-${plugin} PRIVATE user32)
endforeach()
enable_testing()
add_executable(follow-camera-test tests/follow_camera_test.cpp)
target_include_directories(follow-camera-test PRIVATE plugin)
add_test(NAME follow-camera COMMAND follow-camera-test)
