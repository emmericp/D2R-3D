cmake_minimum_required(VERSION 3.20)
project(D2R3DPlugins LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
foreach(plugin IN ITEMS 3dcam renderdistance)
  add_library(d2rl-${plugin} SHARED plugin/${plugin}.cpp plugin/plugin.rc)
  target_include_directories(d2rl-${plugin} PRIVATE ../PluginSDK/include)
  target_compile_options(d2rl-${plugin} PRIVATE /utf-8 /W3 /EHsc /permissive-)
  target_link_libraries(d2rl-${plugin} PRIVATE user32)
endforeach()
add_library(d2rl-sky SHARED plugin/sky.cpp plugin/plugin.rc)
target_include_directories(d2rl-sky PRIVATE ../PluginSDK/include third_party/reshade/include)
target_compile_options(d2rl-sky PRIVATE /utf-8 /W3 /EHsc /permissive-)
target_link_libraries(d2rl-sky PRIVATE user32)
enable_testing()
add_executable(follow-camera-test tests/follow_camera_test.cpp)
target_include_directories(follow-camera-test PRIVATE plugin)
add_test(NAME follow-camera COMMAND follow-camera-test)
add_executable(sky-model-test tests/sky_model_test.cpp)
target_include_directories(sky-model-test PRIVATE plugin)
add_test(NAME sky-model COMMAND sky-model-test)
add_executable(camera-picking-test tests/camera_picking_test.cpp)
target_include_directories(camera-picking-test PRIVATE plugin)
add_test(NAME camera-picking COMMAND camera-picking-test)
add_executable(camera-orbit-test tests/camera_orbit_test.cpp)
target_include_directories(camera-orbit-test PRIVATE plugin)
add_test(NAME camera-orbit COMMAND camera-orbit-test)
add_executable(camera-metadata-test tests/camera_metadata_test.cpp)
target_include_directories(camera-metadata-test PRIVATE plugin)
add_test(NAME camera-metadata COMMAND camera-metadata-test)
add_executable(camera-inspection-test tests/camera_inspection_test.cpp)
target_include_directories(camera-inspection-test PRIVATE plugin)
add_test(NAME camera-inspection COMMAND camera-inspection-test)
